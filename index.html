from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import SQLiteSaver
from langchain_openai import ChatOpenAI
from langchain_community.tools.tavily_search import TavilySearchResults
from langchain.tools import tool
from langchain_core.runnables import RunnableConfig

# Initialize LLM with advanced settings
llm = ChatOpenAI(
    model="gpt-4o",
    temperature=0.6,
    openai_api_key="YOUR_API_KEY",
    max_tokens=1024
)

# Custom tools
@tool
def medical_image_analyzer(query: str) -> str:
    """Analyzes medical imaging data using pre-trained CLIP models."""
    return f"Analyzing {query} with CLIP: Potential findings suggest further review."

@tool
def university_recommender(gpa: float, field: str) -> str:
    """Recommends universities based on GPA and field of study."""
    score = gpa * 100
    return f"Top universities for {field} with GPA {gpa}: Texas A&M, MIT (score: {score})."

tools = [TavilySearchResults(max_results=5), medical_image_analyzer, university_recommender]

# SQLite memory for persistence
memory = SQLiteSaver.from_conn_string(":memory:")

# Create agent with state management
agent_executor = create_react_agent(
    llm,
    tools,
    checkpointer=memory,
    stream_mode="updates"
)

# Example execution with config
config = RunnableConfig(configurable={"thread_id": "thread_1"})
response = agent_executor.invoke(
    {"messages": [{"role": "user", "content": "Recommend universities for CS with GPA 3.7, then analyze a chest X-ray."}]},
    config
)
print("Agent Response:", response["messages"][-1].content)
